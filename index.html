BEDROCK SCRIPT API (@minecraft/server) // LIVE CONTAINER WATCHER

Drop this script into a Bedrock Behavior Pack (scripts/main.js) or Dedicated Server to automatically inspect any Barrel, Double Chest, Ender Chest, Shulker Box, or Hopper when a player interacts with it, fire the in-game /title @a title "ULTIMATE BACKPACKER" celebration on every PC/Console/Mobile screen, and sync directly to /api/backpack-webhook: 

import { world, system } from "@minecraft/server";

const TRACKED_FOODS = new Set([
  "minecraft:spider_eye", "minecraft:sweet_berries", "minecraft:cooked_salmon",
  "minecraft:cooked_beef", "minecraft:golden_carrot", "minecraft:baked_potato",
  "minecraft:bread", "minecraft:cooked_porkchop", "minecraft:golden_apple",
  "minecraft:enchanted_golden_apple", "minecraft:dried_kelp"
]);

world.afterEvents.playerInteractWithBlock.subscribe(({ player, block }) => {
  system.run(() => {
    const inv = block.getComponent("minecraft:inventory")?.container;
    if (!inv) return;
    const totalSlots = inv.size; // 27 (Barrel/Shulker) or 54 (Double Chest) or 5 (Hopper)
    let maxedFoodSlots = 0;
    let totalFoodCount = 0;
    let detectedFoodId = null;

    for (let i = 0; i < totalSlots; i++) {
      const item = inv.getItem(i);
      if (item && TRACKED_FOODS.has(item.typeId)) {
        detectedFoodId = item.typeId;
        totalFoodCount += item.amount;
        if (item.amount === 64) maxedFoodSlots++;
      }
    }

    // Level 1 (Barrel 27x64), Level 2 (Double Chest 54x64), Level 4 (Shulker 27x64)
    if ((totalSlots === 27 || totalSlots === 54) && maxedFoodSlots === totalSlots) {
      const tierName = block.typeId.includes("barrel") ? "LEVEL 1: BARREL MASTER"
        : block.typeId.includes("shulker") ? "LEVEL 4: SHULKER DENSITY"
        : "LEVEL 2: DOUBLE CHEST";
      player.runCommandAsync(`title @a title "§a${tierName}"`);
      player.runCommandAsync(`title @a subtitle "§e${player.name} filled ${totalSlots}x64 (${totalFoodCount} ${detectedFoodId.replace('minecraft:','')})!"`);
      player.runCommandAsync(`playsound random.levelup @a`);
    }

    // Level 5 (Hopper / Dropper Automated Pipeline: 13+ food items moving through)
    if ((block.typeId.includes("hopper") || block.typeId.includes("dropper")) && totalFoodCount >= 13) {
      player.runCommandAsync(`title @a title "§6ULTIMATE BACKPACKER"`);
      player.runCommandAsync(`title @a subtitle "§a${player.name} automated ${totalFoodCount} food items via Hoppers & Droppers!"`);
      player.runCommandAsync(`playsound ui.toast.challenge_complete @a`);
    }
  });
});
